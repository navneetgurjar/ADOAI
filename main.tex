\newcommand{\fullbleedimage}[1]{%
  \AddToShipoutPictureBG*{%
    \AtPageLowerLeft{%
      \makebox(0,0)[bl]{\includegraphics[width=\paperwidth,height=\paperheight]{#1}}%
    }%
  }%
}

% ---------------------------------------------------------------------------
% CHAPTER HEADINGS -- centred, unnumbered (the manuscript already spells the
% chapter number out in words, e.g. "CHAPTER ONE"), with generous drop from
% the top of the page and clear space before the body begins.
% ---------------------------------------------------------------------------
\usepackage{titlesec}
\titleformat{\chapter}[display]
  {\centering\normalfont}
  {}
  {0pt}
  {\Large\scshape}
\titlespacing*{\chapter}{0pt}{0pt}{40pt}

% Custom commands for a chapter's heading block, used by chapter_manifest.tex
% #1 = chapter label (e.g. "CHAPTER ONE"), #2 = character name, #3 = image key
% (pass an empty {} for #3 if the chapter has no image). The \chapter*{}
% page-break happens FIRST, then the image and title are placed together on
% that same fresh page -- this ordering matters, because \chapter*{} forces
% its own page break and would otherwise orphan an image placed before it
% onto a separate page by itself.
\newcommand{\novelchapter}[3]{%
  % Ornament appears at the END of the preceding chapter (before the page break)
  \par\vspace{24pt}
  \begin{center}
    \rule{0.18\textwidth}{0.3pt}\enspace{\large ❦}\enspace\rule{0.18\textwidth}{0.3pt}
  \end{center}
  \vspace{10pt}
  \chapter*{}%
  \vspace*{0.06\textheight}%
  \ifx&#3&\else\chapterimage{#3}\fi
  \vspace{10pt}
  {\centering\Large\scshape #1 \par}%
  \vspace{6pt}
  {\centering\large\itshape #2 \par}%
  \vspace{28pt}%
}
\newcommand{\novelchapterprologue}[2]{%
  \chapter*{}%
  \vspace*{0.06\textheight}%
  \ifx&#2&\else\chapterimage{#2}\fi
  \vspace{10pt}
  {\centering\Large\scshape #1 \par}%
  \vspace{28pt}%
}

% ---------------------------------------------------------------------------
% CHAPTER IMAGE HOOK -- silently does nothing if the file isn't found, so the
% document still compiles cleanly even for chapters with no illustration.
% \chapterimage{photo7}  -->  looks for photo7.png in the working directory.
% ---------------------------------------------------------------------------
\newcommand{\chapterimage}[1]{%
  \IfFileExists{#1.png}{%
    \begin{center}
      \includegraphics[width=0.55\textwidth]{#1.png}
    \end{center}
    \vspace{4pt}
  }{}%
}

% ---------------------------------------------------------------------------
% SCENE BREAK -- minimal short rule for within-chapter scene breaks.
% The ornate fleuron (❦) is reserved for chapter transitions only (see \novelchapter).
% ---------------------------------------------------------------------------
\newcommand{\sceneBreak}{%
  \par\vspace{14pt}
  \begin{center}
    \rule{2cm}{0.3pt}%
  \end{center}
  \vspace{14pt}\par
}
